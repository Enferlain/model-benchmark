const generatePlaceholder = (text, color, pattern = 'grid') => {
  const bg = color === 'blue' ? '#e0f2fe' : color === 'purple' ? '#f3e8ff' : '#ecfdf5';
  const stroke = color === 'blue' ? '#38bdf8' : color === 'purple' ? '#c084fc' : '#34d399';
  
  return (
    <div className={`w-full h-full flex items-center justify-center relative overflow-hidden`} style={{ backgroundColor: bg }}>
      <svg className="absolute inset-0 w-full h-full opacity-20" width="100%" height="100%">
        <defs>
          <pattern id={`pattern-${text}`} x="0" y="0" width="40" height="40" patternUnits="userSpaceOnUse">
             {pattern === 'grid' && <path d="M 40 0 L 0 0 0 40" fill="none" stroke={stroke} strokeWidth="1"/>}
             {pattern === 'dots' && <circle cx="20" cy="20" r="2" fill={stroke} />}
             {pattern === 'waves' && <path d="M 0 20 Q 10 10 20 20 T 40 20" fill="none" stroke={stroke} strokeWidth="2" />}
          </pattern>
        </defs>
        <rect width="100%" height="100%" fill={`url(#pattern-${text})`} />
      </svg>
      <div className="z-10 text-center p-4">
        <div className="text-4xl mb-2">
           {text.includes('Ref') ? '🖼️' : text.includes('A') ? '🤖 A' : '🤖 B'}
        </div>
        <span className="font-bold opacity-60 text-slate-700 uppercase tracking-wider text-sm">{text}</span>
      </div>
    </div>
  );
};

    setIsReferenceExpanded(prev => !prev);

  // Keyboard listener for Spacebar or specific keys
  useEffect(() => {
    const handleKeyDown = (e) => {
      // Space to toggle reference view
      if (e.code === 'Space') {
        e.preventDefault(); // Prevent scrolling
        toggleReference();
      }
      // 1 or ArrowLeft to select Left
      if (e.key === '1' || e.code === 'ArrowLeft') setSelectedImage('left');
      // 2 or ArrowRight to select Right

          {selectedImage === 'right' && (
            <div className="absolute inset-0 z-20 bg-pink-500/10 flex items-center justify-center pointer-events-none animate-in fade-in duration-300">
              <CheckCircle2 className="text-pink-500 w-24 h-24 drop-shadow-lg scale-125" />
            </div>
          )}

           {/* Image Content B */}
           <div className="w-full h-full bg-slate-900 flex items-center justify-center relative overflow-hidden">
             <img 
               src="https://picsum.photos/seed/cyberpunk2/800/1200" 
               alt="Candidate B"
               className="w-full h-full object-cover opacity-90 group-hover:opacity-100 transition-opacity duration-500"
             />
             <div className="absolute bottom-0 inset-x-0 h-32 bg-gradient-to-t from-black/80 to-transparent"></div>

             {/* Text */}
             <div className="absolute bottom-6 right-6 z-10">
                <span className="text-slate-300 font-mono text-sm tracking-widest bg-black/50 px-2 py-1 rounded border border-white/10">CANDIDATE B</span>
             </div>
          </div>
        </div>

      </main>

      {/* Footer controls/status */}
      <footer className="p-4 flex justify-between items-center text-slate-600 text-xs border-t border-slate-800 bg-slate-900/50">
         <div>Lumi's Layout Demo v1.0</div>
         <div className="flex gap-4">
            <span>Status: <span className={selectedImage ? "text-green-400" : "text-yellow-400"}>{selectedImage ? "Voted" : "Waiting"}</span></span>
            <span>Mode: <span className="text-blue-400">{isReferenceExpanded ? "Reference View" : "Standard View"}</span></span>
         </div>
      </footer>

    </div>
  );
};
